# Lumen serverless Telegram webhook entry — BOT_TOKEN from env only.
from __future__ import annotations

import asyncio
import json
import logging
import os
import sys
import types
from http.server import BaseHTTPRequestHandler
from pathlib import Path

logging.basicConfig(level=logging.INFO)
log = logging.getLogger("lumen.serverless.webhook")

_ROOT = Path(__file__).resolve().parent.parent
if str(_ROOT) not in sys.path:
    sys.path.insert(0, str(_ROOT))

_FRAMEWORK = "@@FRAMEWORK@@"
_ENTRY = "@@ENTRY@@"
_APP = None  # cached application / bot / dispatcher bundle


def _token() -> str:
    return (os.environ.get("BOT_TOKEN") or os.environ.get("TELEGRAM_BOT_TOKEN") or "").strip()


def _webhook_secret() -> str:
    return (
        os.environ.get("TELEGRAM_WEBHOOK_SECRET")
        or os.environ.get("WEBHOOK_SECRET")
        or os.environ.get("TBE_HOST_WEBHOOK_SECRET")
        or ""
    ).strip()


def _run_async(coro):
    try:
        return asyncio.run(coro)
    except RuntimeError:
        loop = asyncio.new_event_loop()
        try:
            return loop.run_until_complete(coro)
        finally:
            loop.close()


def _noop(*_a, **_k):
    return None


async def _anop(*_a, **_k):
    return None


def _install_polling_stubs() -> None:
    """Prevent run_polling/start_polling from blocking or starting long-poll on import."""
    try:
        import telegram.ext as ext
        if hasattr(ext, "Application"):
            ext.Application.run_polling = _noop  # type: ignore[method-assign]
            if hasattr(ext.Application, "run_webhook"):
                ext.Application.run_webhook = _noop  # type: ignore[method-assign]
        if hasattr(ext, "Updater"):
            if hasattr(ext.Updater, "start_polling"):
                ext.Updater.start_polling = _noop  # type: ignore[method-assign]
            if hasattr(ext.Updater, "start_webhook"):
                ext.Updater.start_webhook = _noop  # type: ignore[method-assign]
    except Exception:
        pass
    try:
        import telebot
        if hasattr(telebot, "TeleBot"):
            telebot.TeleBot.polling = _noop  # type: ignore[method-assign]
            telebot.TeleBot.infinity_polling = _noop  # type: ignore[method-assign]
    except Exception:
        pass
    try:
        from aiogram import executor  # aiogram v2
        executor.start_polling = _noop  # type: ignore[method-assign]
        if hasattr(executor, "start_webhook"):
            executor.start_webhook = _noop  # type: ignore[method-assign]
    except Exception:
        pass


def _import_entry():
    _install_polling_stubs()
    import importlib
    # Re-import cleanly
    if _ENTRY in sys.modules:
        return sys.modules[_ENTRY]
    return importlib.import_module(_ENTRY)


def _find_ptb_app(mod):
    for name in ("application", "app", "APPLICATION", "APP", "telegram_app"):
        app = getattr(mod, name, None)
        if app is not None and hasattr(app, "process_update"):
            return app
    for name in ("build_application", "create_application", "make_application", "get_application", "build_app"):
        fn = getattr(mod, name, None)
        if callable(fn):
            try:
                app = fn()
            except TypeError:
                tok = _token()
                try:
                    app = fn(tok) if tok else None
                except Exception:
                    app = None
            except Exception:
                app = None
            if app is not None and hasattr(app, "process_update"):
                return app
    # Scan module attrs for Application-like
    for _n, obj in vars(mod).items():
        if obj is not None and hasattr(obj, "process_update") and hasattr(obj, "bot"):
            return obj
    return None


def _find_telebot(mod):
    for name in ("bot", "BOT", "tele_bot", "telegram_bot"):
        bot = getattr(mod, name, None)
        if bot is not None and hasattr(bot, "process_new_updates"):
            return bot
    for _n, obj in vars(mod).items():
        if obj is not None and hasattr(obj, "process_new_updates") and hasattr(obj, "get_me"):
            return obj
    return None


def _find_aiogram(mod):
    dp = None
    bot = None
    for name in ("dp", "dispatcher", "Dispatcher", "dp_"):
        dp = getattr(mod, name, None)
        if dp is not None:
            break
    for name in ("bot", "Bot", "BOT"):
        bot = getattr(mod, name, None)
        if bot is not None:
            break
    if dp is None:
        for _n, obj in vars(mod).items():
            if obj is not None and (hasattr(obj, "feed_update") or hasattr(obj, "process_update")):
                dp = obj
                break
    return dp, bot


def _ensure_runtime():
    global _APP
    if _APP is not None:
        return _APP
    mod = _import_entry()
    bundle = {"module": mod, "framework": _FRAMEWORK}
    if _FRAMEWORK == "python-telegram-bot":
        bundle["app"] = _find_ptb_app(mod)
    elif _FRAMEWORK == "telebot":
        bundle["bot"] = _find_telebot(mod)
    elif _FRAMEWORK == "aiogram":
        dp, bot = _find_aiogram(mod)
        bundle["dp"] = dp
        bundle["bot"] = bot
    else:
        bundle["app"] = _find_ptb_app(mod)
        if bundle["app"] is None:
            bundle["bot"] = _find_telebot(mod)
        if bundle.get("app") is None and bundle.get("bot") is None:
            dp, bot = _find_aiogram(mod)
            bundle["dp"] = dp
            bundle["bot"] = bot
    _APP = bundle
    return bundle


async def _ptb_process(app, update_data: dict):
    from telegram import Update
    bot = getattr(app, "bot", None)
    update = Update.de_json(update_data, bot)
    # Initialize + start once (PTB v20+ needs both for handlers to reply)
    if not getattr(app, "_lumen_inited", False):
        if hasattr(app, "initialize"):
            await app.initialize()
        if hasattr(app, "start"):
            try:
                await app.start()
            except Exception:
                pass
        app._lumen_inited = True  # type: ignore[attr-defined]
    result = app.process_update(update)
    if hasattr(result, "__await__"):
        await result


def process_telegram_update(update_data: dict) -> None:
    if not _token():
        raise RuntimeError("bot_token_missing_in_env")
    if not isinstance(update_data, dict):
        raise RuntimeError("invalid_update")
    bundle = _ensure_runtime()
    fw = bundle.get("framework") or _FRAMEWORK

    if fw == "python-telegram-bot" or bundle.get("app") is not None:
        app = bundle.get("app")
        if app is None:
            raise RuntimeError("ptb_application_not_found")
        _run_async(_ptb_process(app, update_data))
        return

    if fw == "telebot" or bundle.get("bot") is not None and hasattr(bundle.get("bot"), "process_new_updates"):
        bot = bundle.get("bot")
        if bot is None:
            raise RuntimeError("telebot_instance_not_found")
        from telebot.types import Update
        upd = Update.de_json(update_data)
        bot.process_new_updates([upd])
        return

    if fw == "aiogram" or bundle.get("dp") is not None:
        from aiogram.types import Update
        dp = bundle.get("dp")
        bot = bundle.get("bot")
        if dp is None:
            raise RuntimeError("aiogram_dispatcher_not_found")
        if bot is None:
            from aiogram import Bot
            bot = Bot(token=_token())
            bundle["bot"] = bot
        update = Update(**update_data)

        async def _run():
            if hasattr(dp, "feed_update"):
                await dp.feed_update(bot, update)
            elif hasattr(dp, "process_update"):
                await dp.process_update(update)
            else:
                raise RuntimeError("aiogram_no_feed")
        _run_async(_run())
        return

    raise RuntimeError("no_bot_runtime_found")


def _check_secret(headers) -> bool:
    expected = _webhook_secret()
    if not expected:
        return True  # not configured — allow (register path may omit)
    got = ""
    try:
        got = headers.get("X-Telegram-Bot-Api-Secret-Token") or headers.get("x-telegram-bot-api-secret-token") or ""
    except Exception:
        got = ""
    if not got:
        return False
    if len(got) != len(expected):
        return False
    # constant-time
    res = 0
    for a, b in zip(got.encode(), expected.encode()):
        res |= a ^ b
    return res == 0


class handler(BaseHTTPRequestHandler):
    def log_message(self, fmt, *args):
        log.info(fmt, *args)

    def _send(self, code: int, body: dict):
        raw = json.dumps(body).encode("utf-8")
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(raw)))
        self.end_headers()
        self.wfile.write(raw)

    def do_GET(self):
        self._send(
            200,
            {
                "ok": True,
                "service": "lumen-bot",
                "framework": _FRAMEWORK,
                "entry": _ENTRY,
                "token_configured": bool(_token()),
            },
        )

    def do_POST(self):
        if not _check_secret(self.headers):
            self._send(403, {"ok": False, "error": "forbidden"})
            return
        try:
            length = int(self.headers.get("Content-Length") or 0)
        except Exception:
            length = 0
        if length <= 0 or length > 2_000_000:
            self._send(400, {"ok": False, "error": "invalid_body"})
            return
        raw = self.rfile.read(length)
        try:
            data = json.loads(raw.decode("utf-8"))
        except Exception:
            self._send(400, {"ok": False, "error": "invalid_json"})
            return
        if not isinstance(data, dict):
            self._send(400, {"ok": False, "error": "invalid_update"})
            return
        try:
            process_telegram_update(data)
        except Exception as exc:
            log.exception("update_failed")
            self._send(500, {"ok": False, "error": type(exc).__name__})
            return
        self._send(200, {"ok": True})
